class OccEventEnum:
    """Enumerate symmetrically distinct OccEvent for a given Prim

    OccEventEnum enumerates canonical, symmetrically distinct
    :class:`~libcasm.occ_events.OccEvent` for a given
    :class:`~libcasm.configuration.Prim`, optionally adding them to an
    :class:`~libcasm.enumerate.OccEventSet`.

    Example usage::

        import libcasm.configuration as casmconfig
        import libcasm.xtal.prims as xtal_prims
        from libcasm.enumerate import OccEventEnum, OccEventSet

        xtal_prim = xtal_prims.FCC(r=1.0, occ_dof=["A", "Va"])
        prim = casmconfig.Prim(xtal_prim)
        event_set = OccEventSet(prim=prim)

        enumerator = OccEventEnum(prim=prim, event_set=event_set)
        for i, event in enumerate(enumerator.by_n_orbits(n_orbits=[0, 0, 3])):
            print(f"Event {i}:")
            event_set.print(event, coordinate_mode="cart")
            print()
        print("# of events =", len(event_set))

    """

    def __init__(
        self,
        prim: casmconfig.Prim,
        system: Optional[OccSystem] = None,
        event_set: Optional[OccEventSet] = None,
    ):
        """
        Parameters
        ----------
        prim: ~libcasm.configuration.Prim
            The primitive crystal structure.
        system: Optional[~libcasm.occ_events.OccSystem] = None
            An :class:`~libcasm.occ_events.OccSystem`, used for indexing
            occupations. If None, one is constructed from ``prim``. Ignored
            if ``event_set`` is provided; in that case the system from
            ``event_set`` is used.
        event_set: Optional[~libcasm.enumerate.OccEventSet] = None
            If provided, enumerated events are added to this set. Each yielded
            event is added before being yielded. If provided, ``prim`` must be
            the same object as ``event_set.prim``, and ``system`` is ignored in
            favour of ``event_set.event_system``.

        Raises
        ------
        ValueError
            If ``event_set`` is provided and ``event_set.prim is not prim``.
        """
        if event_set is not None and event_set.prim is not prim:
            raise ValueError("OccEventEnum: event_set.prim is not prim")
        self._prim = prim
        if event_set is not None:
            system = event_set.event_system
        elif system is None:
            system = OccSystem(xtal_prim=prim.xtal_prim)
        self._system = system
        self._event_set = event_set

        # State set during enumeration
        self._cluster_specs = None
        self._enum_index = None

    @property
    def prim(self) -> casmconfig.Prim:
        """The primitive crystal structure."""
        return self._prim

    @property
    def system(self) -> OccSystem:
        """The :class:`~libcasm.occ_events.OccSystem` used for indexing."""
        return self._system

    @property
    def event_set(self) -> Optional[OccEventSet]:
        """The :class:`~libcasm.enumerate.OccEventSet`, if provided."""
        return self._event_set

    @property
    def cluster_specs(self) -> Optional[clust.ClusterSpecs]:
        """During enumeration, the :class:`~libcasm.clusterography.ClusterSpecs`
        used to generate cluster orbits."""
        return self._cluster_specs

    @property
    def enum_index(self) -> Optional[int]:
        """During enumeration, the index of the current event starting from 0."""
        return self._enum_index

    def by_n_orbits(
        self,
        n_orbits: list[int],
        occevent_counter_params: dict = {},
        custom_events: list[OccEvent] = [],
    ):
        """Enumerate OccEvent using the first N cluster orbits of each size.

        Constructs :class:`~libcasm.clusterography.ClusterSpecs` using
        :func:`~libcasm.enumerate.make_first_n_orbits_cluster_specs` and then
        calls :func:`~libcasm.occ_events.make_canonical_prim_periodic_occevents`
        to enumerate symmetrically distinct :class:`~libcasm.occ_events.OccEvent`.

        Parameters
        ----------
        n_orbits: list[int]
            Number of orbits to include for each branch (cluster size).
            ``n_orbits[b]`` is the number of orbits for clusters of size ``b``.
            For example, ``n_orbits=[0, 0, 3]`` includes the first 3 pair-cluster
            orbits (branch 2).
        occevent_counter_params: dict = {}
            Options filtering which OccEvent are generated. See
            :func:`~libcasm.occ_events.make_canonical_prim_periodic_occevents`
            for the full list of options.
        custom_events: list[~libcasm.occ_events.OccEvent] = []
            OccEvent that should be included in the results regardless of the
            other options.

        Yields
        ------
        event: ~libcasm.occ_events.OccEvent
            Symmetrically distinct OccEvent, one at a time.
        """
        self._cluster_specs = make_first_n_orbits_cluster_specs(
            prim=self._prim,
            n_orbits=n_orbits,
        )
        if self._enum_index is None:
            self._enum_index = 0
        events = make_canonical_prim_periodic_occevents(
            system=self._system,
            cluster_specs=self._cluster_specs,
            occevent_counter_params=occevent_counter_params,
            custom_events=custom_events,
        )
        for event in events:
            if self._event_set is not None:
                self._event_set.add_event(event)
            yield event
            self._enum_index += 1

    def by_max_length(
        self,
        max_length: list[float],
        occevent_counter_params: dict = {},
        custom_events: list[OccEvent] = [],
    ):
        """Enumerate OccEvent using clusters up to a maximum length.

        Constructs :class:`~libcasm.clusterography.ClusterSpecs` directly from
        ``max_length`` and then calls
        :func:`~libcasm.occ_events.make_canonical_prim_periodic_occevents`
        to enumerate symmetrically distinct :class:`~libcasm.occ_events.OccEvent`.

        Parameters
        ----------
        max_length: list[float]
            Maximum cluster site-to-site distance for each branch (cluster size).
            ``max_length[b]`` is the maximum distance for clusters of size ``b``.
            For example, ``max_length=[0., 0., 4.0]`` includes all pair clusters
            (branch 2) with site-to-site distance up to 4.0.
        occevent_counter_params: dict = {}
            Options filtering which OccEvent are generated. See
            :func:`~libcasm.occ_events.make_canonical_prim_periodic_occevents`
            for the full list of options.
        custom_events: list[~libcasm.occ_events.OccEvent] = []
            OccEvent that should be included in the results regardless of the
            other options.

        Yields
        ------
        event: ~libcasm.occ_events.OccEvent
            Symmetrically distinct OccEvent, one at a time.
        """
        self._cluster_specs = clust.ClusterSpecs(
            xtal_prim=self._prim.xtal_prim,
            generating_group=sym_info.make_factor_group(self._prim.xtal_prim),
            max_length=max_length,
        )
        if self._enum_index is None:
            self._enum_index = 0
        events = make_canonical_prim_periodic_occevents(
            system=self._system,
            cluster_specs=self._cluster_specs,
            occevent_counter_params=occevent_counter_params,
            custom_events=custom_events,
        )
        for event in events:
            if self._event_set is not None:
                self._event_set.add_event(event)
            yield event
            self._enum_index += 1

    def by_cluster_list(
        self,
        clusters: list[clust.Cluster],
        occevent_counter_params: dict = {},
        custom_events: list[OccEvent] = [],
    ):
        """Enumerate OccEvent using an explicitly provided list of clusters.

        Builds a :class:`~libcasm.clusterography.ClusterSpecs` from the provided
        clusters (one :class:`~libcasm.clusterography.ClusterOrbitGenerator` per
        unique cluster prototype) and then calls
        :func:`~libcasm.occ_events.make_canonical_prim_periodic_occevents`.

        Parameters
        ----------
        clusters: list[~libcasm.clusterography.Cluster]
            Cluster prototypes specifying which orbits to enumerate over.
            Duplicate orbits (same prototype up to the factor group) are
            silently ignored.
        occevent_counter_params: dict = {}
            Options filtering which OccEvent are generated. See
            :func:`~libcasm.occ_events.make_canonical_prim_periodic_occevents`
            for the full list of options.
        custom_events: list[~libcasm.occ_events.OccEvent] = []
            OccEvent that should be included in the results regardless of the
            other options.

        Yields
        ------
        event: ~libcasm.occ_events.OccEvent
            Symmetrically distinct OccEvent, one at a time.
        """
        custom_generators = [
            clust.ClusterOrbitGenerator(
                prototype=cluster,
                include_subclusters=False,
            )
            for cluster in clusters
        ]
        self._cluster_specs = clust.ClusterSpecs(
            xtal_prim=self._prim.xtal_prim,
            generating_group=self._prim.factor_group,
            custom_generators=custom_generators,
        )
        if self._enum_index is None:
            self._enum_index = 0
        events = make_canonical_prim_periodic_occevents(
            system=self._system,
            cluster_specs=self._cluster_specs,
            occevent_counter_params=occevent_counter_params,
            custom_events=custom_events,
        )
        for event in events:
            if self._event_set is not None:
                self._event_set.add_event(event)
            yield event
            self._enum_index += 1

    def by_branch_and_orbit_index(
        self,
        orbits: list[tuple[int, int]],
        occevent_counter_params: dict = {},
        custom_events: list[OccEvent] = [],
    ):
        """Enumerate OccEvent for specific cluster orbits by (branch, orbit) index.

        Selects orbits by branch (cluster size) and 1-based orbit index within
        that branch. Builds a :class:`~libcasm.clusterography.ClusterSpecs` from
        the selected orbit prototypes and calls
        :func:`~libcasm.occ_events.make_canonical_prim_periodic_occevents`.

        Parameters
        ----------
        orbits: list[tuple[int, int]]
            Each element is a ``(branch, orbit_index)`` pair, where ``branch``
            is the cluster size (number of sites) and ``orbit_index`` is the
            1-based index of the orbit within that branch. A single tuple may
            also be passed and will be treated as a one-element list.
        occevent_counter_params: dict = {}
            Options filtering which OccEvent are generated. See
            :func:`~libcasm.occ_events.make_canonical_prim_periodic_occevents`
            for the full list of options.
        custom_events: list[~libcasm.occ_events.OccEvent] = []
            OccEvent that should be included in the results regardless of the
            other options.

        Yields
        ------
        event: ~libcasm.occ_events.OccEvent
            Symmetrically distinct OccEvent, one at a time.
        """
        # Normalise: allow a bare tuple as a single-element input
        if isinstance(orbits, tuple):
            orbits = [orbits]

        # Determine how many orbits are needed per branch
        n_orbits_needed: dict[int, int] = {}
        for branch, orbit_idx in orbits:
            if orbit_idx < 1:
                raise ValueError(f"orbit_index must be >= 1, got {orbit_idx}")
            n_orbits_needed[branch] = max(n_orbits_needed.get(branch, 0), orbit_idx)

        max_branch = max(n_orbits_needed)
        n_orbits = [0] * (max_branch + 1)
        for branch, count in n_orbits_needed.items():
            n_orbits[branch] = count

        # Generate enough orbits to cover the requested indices
        all_orbits = make_first_n_orbits(prim=self._prim, n_orbits=n_orbits)

        # Group orbits by branch
        by_branch: dict[int, list] = {}
        for orbit in all_orbits:
            if len(orbit) == 0:
                continue
            b = len(orbit[0])
            by_branch.setdefault(b, []).append(orbit)

        # Build one ClusterOrbitGenerator per requested (branch, orbit_idx)
        custom_generators = []
        for branch, orbit_idx in orbits:
            cluster = by_branch[branch][orbit_idx - 1][0]
            custom_generators.append(
                clust.ClusterOrbitGenerator(
                    prototype=cluster,
                    include_subclusters=False,
                )
            )

        self._cluster_specs = clust.ClusterSpecs(
            xtal_prim=self._prim.xtal_prim,
            generating_group=self._prim.factor_group,
            custom_generators=custom_generators,
        )
        if self._enum_index is None:
            self._enum_index = 0
        events = make_canonical_prim_periodic_occevents(
            system=self._system,
            cluster_specs=self._cluster_specs,
            occevent_counter_params=occevent_counter_params,
            custom_events=custom_events,
        )
        for event in events:
            if self._event_set is not None:
                self._event_set.add_event(event)
            yield event
            self._enum_index += 1
