def _get_event_occ_names(
    prim: Prim,
    occ_event: OccEvent,
):
    occ_dof = prim.xtal_prim.occ_dof()
    cluster = occ_event.cluster()
    init = occ_event.initial_occupation()
    names = []
    for i, s in enumerate(init):
        site = cluster[i]
        b = site.sublattice()
        names.append(occ_dof[b][s])
    return sorted(names)


def _find_event_orbit_index_in_branch(
    prim: Prim,
    occ_event: OccEvent,
    n_orbits: list[int],
):
    event_cluster = occ_event.cluster().sorted()
    if len(event_cluster):
        event_cluster -= event_cluster[0].unitcell()

    while len(n_orbits) <= len(event_cluster):
        n_orbits.append(0)

    if n_orbits[len(event_cluster)] == 0:
        n_orbits[len(event_cluster)] = 1

    for i in range(len(n_orbits)):
        if i > len(event_cluster):
            n_orbits[i] = 0

    cluster_specs = make_first_n_orbits_cluster_specs(
        prim=prim,
        n_orbits=n_orbits,
    )
    i_orbit_in_branch = None
    total_orbits = 0
    while i_orbit_in_branch is None:
        orbits = cluster_specs.make_orbits()
        if len(orbits) <= total_orbits:
            raise Exception(
                "Error in _find_event_orbit_index_in_branch: failed adding orbits"
            )
        i_orbit_in_branch_counter = 0
        for i, orbit in enumerate(orbits):
            if len(orbit) == 0:
                continue
            if len(event_cluster) != len(orbit[0]):
                continue
            if event_cluster in orbit:
                i_orbit_in_branch = i_orbit_in_branch_counter
                break
            i_orbit_in_branch_counter += 1
        if i_orbit_in_branch is None:
            n_orbits[len(event_cluster)] += 1
            cluster_specs = make_first_n_orbits_cluster_specs(
                prim=prim,
                n_orbits=n_orbits,
            )

    return i_orbit_in_branch_counter

class OccEventSet:
    """Data structure for holding unique OccEvent objects.

    OccEvent objects are stored as :class:`OccEventRecord`, each of which
    contains the event and event name which is a combination of the orbit branch
    (number of sites) for the underlying cluster, orbit index (beginning at 1), and a
    unique event_id assigned within the same (branch, orbit) pair.

    .. rubric:: Special Methods

    - ``for record in event_set``: Iterate over :class:`OccEventRecord` in
      sorted order.
    - ``if event in event_set``: Check if an :class:`OccEvent` is in the set.
    - ``if event_name in event_set``: Check if an event name (str) is in the
      set.
    - ``len(event_set)``: Return the number of records in the set.
    """

    def __init__(
        self,
        prim: Prim,
        system: Optional[OccSystem] = None,
    ):
        self.prim = prim

        if system is None:
            self.event_system = OccSystem(xtal_prim=prim.xtal_prim)
        else:
            self.event_system = system

        self._records = sortedcontainers.SortedList()
        """SortedList[OccEventRecord]: Sorted list of event records."""

        self._name_to_record = {}
        """dict[str, OccEventRecord]: Mapping of event_name to record."""

        self._next_event_id = sortedcontainers.SortedDict()
        """SortedDict[tuple[int,int],int]: Mapping of (branch, orbit) to the
        next event_id integer to assign."""

        self._n_orbits = []
        """list[int]: Cumulative n_orbits tracking the minimum number of orbits
        per branch needed to index all events added so far. Used as the
        `n_orbits` argument to :func:`make_first_n_orbits_cluster_specs`."""

        self._orbits = []
        """list: Cached result of
        ``make_first_n_orbits_cluster_specs(prim, _n_orbits).make_orbits()``,
        kept in sync with :attr:`_n_orbits`."""

    def clear(self):
        """Remove all records from the set."""
        self._records.clear()
        self._name_to_record.clear()
        self._next_event_id.clear()
        self._n_orbits = []
        self._orbits = []



    def _find_by_event(self, event: OccEvent) -> Optional[OccEventRecord]:
        for record in self._records:
            if record.event == event:
                return record
        return None

    def _find_orbit_for_event(self, event: OccEvent) -> int:
        """Return the 1-based orbit index for an event within its branch.

        Updates :attr:`_n_orbits` and :attr:`_orbits` as needed to encompass
        the event's orbit.

        Parameters
        ----------
        event : OccEvent

        Returns
        -------
        orbit : int
            1-based orbit index within the branch.
        """
        event_cluster = event.cluster().sorted()
        if len(event_cluster):
            event_cluster -= event_cluster[0].unitcell()

        branch = len(event_cluster)

        # Ensure _n_orbits covers this branch with at least 1 orbit.
        while len(self._n_orbits) <= branch:
            self._n_orbits.append(0)
        if self._n_orbits[branch] == 0:
            self._n_orbits[branch] = 1

        # Use a working copy truncated to this branch so the search does not
        # regenerate higher-branch orbits or clobber their stored counts.
        working = list(self._n_orbits[: branch + 1])

        cluster_specs = make_first_n_orbits_cluster_specs(
            prim=self.prim,
            n_orbits=working,
        )
        orbits = cluster_specs.make_orbits()

        i_orbit_in_branch = None
        while i_orbit_in_branch is None:
            i_orbit_in_branch_counter = 0
            for orbit in orbits:
                if len(orbit) == 0:
                    continue
                if branch != len(orbit[0]):
                    continue
                if event_cluster in orbit:
                    i_orbit_in_branch = i_orbit_in_branch_counter
                    break
                i_orbit_in_branch_counter += 1
            if i_orbit_in_branch is None:
                working[branch] += 1
                cluster_specs = make_first_n_orbits_cluster_specs(
                    prim=self.prim,
                    n_orbits=working,
                )
                orbits = cluster_specs.make_orbits()

        # Count actual orbits per branch generated by the final cluster specs.
        # make_first_n_orbits_cluster_specs may increase lower-branch orbit
        # counts beyond what was requested, so reading the actual counts back
        # is more accurate than persisting working[branch] directly.
        actual_counts: dict[int, int] = {}
        for orbit in orbits:
            if len(orbit) == 0:
                continue
            b = len(orbit[0])
            actual_counts[b] = actual_counts.get(b, 0) + 1

        # Merge into _n_orbits, only ever increasing (never reducing).
        for b, count in actual_counts.items():
            while len(self._n_orbits) <= b:
                self._n_orbits.append(0)
            self._n_orbits[b] = max(self._n_orbits[b], count)

        # Refresh _orbits from the full (possibly updated) _n_orbits.
        cluster_specs = make_first_n_orbits_cluster_specs(
            prim=self.prim,
            n_orbits=self._n_orbits,
        )
        self._orbits = cluster_specs.make_orbits()

        return i_orbit_in_branch + 1  # convert to 1-based

    def add_event(
        self,
        event: OccEvent,
    ) -> OccEventRecord:
        """Add an OccEvent to the set, auto-assigning branch, orbit, and event_id.

        The orbit index is determined by searching cluster orbits via
        :attr:`_n_orbits` and :attr:`_orbits`, which are updated as needed.
        If an equivalent event is already present, the existing record is
        returned without modification.

        Parameters
        ----------
        event : OccEvent
            The event to add.

        Returns
        -------
        record : OccEventRecord
            The record held by the set.
        """
        existing = self._find_by_event(event)
        if existing is not None:
            return existing
        branch = len(event.cluster())
        orbit = self._find_orbit_for_event(event)
        key = (branch, orbit)
        next_id = self._next_event_id.get(key, 0)
        self._next_event_id[key] = next_id + 1
        record = OccEventRecord(
            event=event,
            system=self.event_system,
            branch=branch,
            orbit=orbit,
            event_id=str(next_id),
        )
        self._records.add(record)
        self._name_to_record[record.event_name] = record
        return record

    def to_dict(self) -> dict:
        """Represent the OccEventSet as a Python dict.

        Persists :attr:`event_system`, :attr:`_n_orbits`,
        :attr:`_next_event_id`, and the event records. The derived members
        :attr:`_orbits` and :attr:`_name_to_record` are not stored; they are
        rebuilt on deserialization.

        Returns
        -------
        data : dict
        """
        records = []
        for record in self._records:
            records.append(
                {
                    "event": record.event.to_dict(system=self.event_system),
                    "branch": record.branch,
                    "orbit": record.orbit,
                    "event_id": record.event_id,
                }
            )

        next_event_id = []
        for (branch, orbit), next_id in self._next_event_id.items():
            next_event_id.append(
                {
                    "branch": branch,
                    "orbit": orbit,
                    "next_id": next_id,
                }
            )

        return {
            "event_system": self.event_system.to_dict(),
            "n_orbits": list(self._n_orbits),
            "next_event_id": next_event_id,
            "records": records,
        }

    @staticmethod
    def from_dict(data: dict, prim: Prim) -> OccEventSet:
        """Construct an OccEventSet from a Python dict.

        Parameters
        ----------
        data : dict
            A dict produced by :meth:`to_dict`.
        prim : libcasm.configuration.Prim
            The :class:`~libcasm.configuration.Prim` for the system.

        Returns
        -------
        event_set : OccEventSet
        """
        event_system = OccSystem.from_dict(
            data=data["event_system"],
            xtal_prim=prim.xtal_prim,
        )
        event_set = OccEventSet(prim=prim, system=event_system)

        # Restore _n_orbits and rebuild _orbits.
        event_set._n_orbits = list(data["n_orbits"])
        if event_set._n_orbits:
            cluster_specs = make_first_n_orbits_cluster_specs(
                prim=prim,
                n_orbits=event_set._n_orbits,
            )
            event_set._orbits = cluster_specs.make_orbits()

        # Restore _next_event_id.
        for entry in data["next_event_id"]:
            key = (entry["branch"], entry["orbit"])
            event_set._next_event_id[key] = entry["next_id"]

        # Restore records directly, bypassing orbit-finding logic.
        for record_data in data["records"]:
            event = OccEvent.from_dict(
                data=record_data["event"],
                system=event_system,
            )
            record = OccEventRecord(
                event=event,
                system=event_system,
                branch=record_data["branch"],
                orbit=record_data["orbit"],
                event_id=record_data["event_id"],
            )
            event_set._records.add(record)
            event_set._name_to_record[record.event_name] = record

        return event_set

            If True, print single atom occupant trajectories using the molecule

            f=f,
            system=self.event_system,
            coordinate_mode=coordinate_mode,
            single_atom_occupant_as_mol=single_atom_occupant_as_mol,
            prec=prec,
        )
        printer(event)
