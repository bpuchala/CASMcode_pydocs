def make_little_group(
    kpoint_cart: np.ndarray,
    prim: Prim,
):
    r"""Make the little group for a k-point.

    The little group is the subgroup of the space group that leaves the k-point
    invariant up to a reciprocal lattice vector.

    As with the prim and supercell factor group, the resulting `little_group` is a
    :py:class:`SymGroup` whose elements are representatives the infinite set of space
    group operations that leave the k-point invariant, (i.e. it includes only
    :math:`\{R_i|t_i\}` to stand for all :math:`\{R_i|t_i + L \cdot n\}` where
    :math:`L` is the prim lattice and :math:`n` is any integer vector), so that group
    multiplication is defined modulo primitive lattice translations.

    Parameters
    ----------
    kpoint_cart: np.ndarray
        A shape (3,) array giving the Cartesian coordinates of the k-point in question.
    prim: Prim
        The prim for which the little group will be constructed.

    Returns
    -------
    little_group: SymGroup
        The subgroup of `prim.factor_group` that leaves the k-point invariant up
        to a reciprocal lattice vector. The head group of `little_group` is set to
        the `prim.factor_group`.
    """
    little_group_indices = set()
    recip_lat = prim.xtal_prim.lattice().reciprocal()
    prim_fg = prim.factor_group
    for prim_fg_index, op in enumerate(prim_fg.elements):
        d = min_periodic_displacement(
            lattice=recip_lat,
            r1=kpoint_cart,
            r2=op.matrix() @ kpoint_cart,
        )
        if np.linalg.norm(d) < recip_lat.tol():
            little_group_indices.add(prim_fg_index)
    return prim_fg.make_subgroup(head_group_index=little_group_indices)

def make_little_group_for_supercell(
    kpoint_cart: np.ndarray,
    supercell: Supercell,
):
    r"""Make the little group for a k-point in a supercell.

    The little group is the subgroup of the space group that leaves the k-point
    invariant up to a reciprocal lattice vector. Here we further restrict to elements
    that leave the supercell lattice vectors invariant.

    As with the prim and supercell factor group, the resulting `little_group` is a
    :py:class:`SymGroup` whose elements are representatives the infinite set of space
    group operations that leave the k-point invariant, (i.e. it includes only
    :math:`\{R_i|t_i\}` to stand for all :math:`\{R_i|t_i + L \cdot n\}` where
    :math:`L` is the prim lattice and :math:`n` is any integer vector), so that group
    multiplication is defined modulo primitive lattice translations.

    Parameters
    ----------
    kpoint_cart: np.ndarray
        A shape (3,) array giving the Cartesian coordinates of the k-point in question.
    supercell: Supercell
        A supercell in which symmetry-adapted basis functions for the k-point will be
        constructed.

    Returns
    -------
    little_group: SymGroup
        The subgroup of `supercell.factor_group` that leaves the k-point invariant up
        to a reciprocal lattice vector. The head group of `little_group` is set to
        the `supercell.prim.factor_group`.
    """
    little_group_indices = set()
    recip_lat = supercell.prim.xtal_prim.lattice().reciprocal()
    prim_fg = supercell.prim.factor_group
    scel_fg = supercell.factor_group
    for i, op in enumerate(scel_fg.elements):
        prim_fg_index = scel_fg.head_group_index[i]
        d = min_periodic_displacement(
            lattice=recip_lat,
            r1=kpoint_cart,
            r2=op.matrix() @ kpoint_cart,
        )
        if np.linalg.norm(d) < recip_lat.tol():
            little_group_indices.add(prim_fg_index)
    return prim_fg.make_subgroup(head_group_index=little_group_indices)

def make_factor_system_for_little_group(
    kpoint_cart: np.ndarray,
    little_group: SymGroup,
):
    r"""Make the factor system for a little group.

    The factor system is a function :math:`\mu(H_i, H_j)` that gives the factor
    relating the product of two group representations to the representation of
    their product,

    .. math::

        \Delta(H_i) \Delta(H_j) = \mu(H_i, H_j) \Delta(H_i H_j),

    where:

    - :math:`H_i` and :math:`H_j` are elements of the group, and
    - :math:`\Delta(H)` is the representation of group element H.

    For a little group of a k-point, the factor system is given by:

    .. math::

        \mu(H_i, H_j) = \exp(-i g_i \cdot t_j)

    where:

    - :math:`k` is the Cartesian coordinate of the k-point in question,
    - :math:`t_j` is the fractional translation associated with :math:`H_j`, and
    - :math:`g_i` is a reciprocal lattice vector that satisfies
      :math:`R_i^{-1} k = k + g_i`, :math:`R_i` being the point group operation
      associated with :math:`H_i`.

    A value other than 1 for :math:`\mu(H_i, H_j)` arises when the little group
    elements include fractional translations (screw axes or glide planes) and
    :math:`g_i \neq \vec{0}`.

    Refer to Bradley and Cracknell, Sec. 3.7, for more details.

    Parameters
    ----------
    kpoint_cart: np.ndarray
        A shape (3,) array giving the Cartesian coordinates of the k-point in question.
    little_group: SymGroup
        Representatives of the little group elements, with point group operations and
        fractional translations. For symmetry adapted basis functions in a supercell,
        this would be a subgroup of the supercell factor group.

    Returns
    -------
    factor_system: np.ndarray[np.complex128[h, h]]
        A shape :math:`(h, h)` array of complex values giving the factor system for the
        little group, where :math:`h` is the number of elements in `little_group`.

    """
    d = len(little_group.elements)
    M = np.zeros((d, d), dtype=complex)
    for i, op_i in enumerate(little_group.elements):
        R_i = op_i.matrix()
        g_i = R_i.T @ kpoint_cart - kpoint_cart
        for j, op_j in enumerate(little_group.elements):
            t_j = op_j.translation()
            M[i, j] = np.exp(-1j * g_i @ t_j)
    return M
