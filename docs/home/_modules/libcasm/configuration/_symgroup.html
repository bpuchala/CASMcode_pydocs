"""Implementations of .symgroup() methods."""

import numpy as np


def _apply_supercell_symop_to_occ_event(op, occ_event, prim):
    """Apply a SupercellSymOp to an OccEvent (rotation + translation).

    Parameters
    ----------
    op : SupercellSymOp
    occ_event : libcasm.occ_events.OccEvent
    prim : Prim

    Returns
    -------
    transformed : libcasm.occ_events.OccEvent
    """
    import libcasm.occ_events as occ

    i_fg = op.prim_factor_group_index()
    rep = occ.OccEventRep(
        integral_site_coordinate_rep=prim.integral_site_coordinate_symgroup_rep[i_fg],
        occupant_rep=prim.occ_symgroup_rep[i_fg],
        atom_position_rep=prim.atom_position_symgroup_rep[i_fg],
    )
    return rep * occ_event + op.translation_frac()


def _supercell_symgroup_impl(ops, objs, supercell, local=None):
    """Refine a list of SupercellSymOp by sequential invariance constraints,
    then return a SymGroup.

    Parameters
    ----------
    ops : list[SupercellSymOp]
        Starting set of operations.
    objs : tuple
        Sequence of objects which should remain invariant under the action of the
        resulting SymGroup, using periodic symmetry (symmetry elements must map the
        objects modulo supercell lattice translations). Supported types:

        - :class:`~libcasm.configuration.Configuration`: uses
          :func:`~libcasm.configuration.make_invariant_subgroup`
        - :class:`~libcasm.clusterography.Cluster`: converts cluster sites
          to linear site indices then uses
          :func:`~libcasm.configuration.make_invariant_subgroup`
        - :class:`~libcasm.occ_events.OccEvent`: filters SupercellSymOp to
          those whose image of the OccEvent equals the original after
          standardization (permutation/reversal), modulo supercell lattice translations
        - Any object with a ``make_invariant_subgroup(group: list[SupercellSymOp])``
          method: calls ``obj.make_invariant_subgroup(group=ops)`` to filter the current
          list of SupercellSymOp.

    supercell : Supercell
        The supercell, used for site index conversion and prim access.
    local : None, Cluster, or tuple[Cluster, ...]
        Optional local clusters that should also remain invariant under the action
        of the resulting SymGroup. Requires the last element of ``objs`` to be a
        :class:`~libcasm.clusterography.Cluster` or
        :class:`~libcasm.occ_events.OccEvent`, which is used as the phenomenal
        cluster/event.

    Returns
    -------
    group : libcasm.sym_info.SymGroup
        A SymGroup with the supercell lattice. If the last element of ``objs`` is a
        Cluster or OccEvent, :func:`~libcasm.clusterography.make_cluster_group`
        or :func:`~libcasm.occ_events.make_occevent_group` is called on the
        SymGroup to fix translations so the phenomenal cluster/event maps
        exactly. If `local` is not None,
        :func:`~libcasm.clusterography.make_local_cluster_group` is applied for each
        local cluster.
    """
    import libcasm.clusterography as clust
    import libcasm.occ_events as occ

    from ._configuration import Configuration, make_invariant_subgroup, make_symgroup

    if not isinstance(objs, tuple):
        raise TypeError("Error in _supercell_symgroup_impl: objs must be a tuple")

    if len(objs) == 0:
        raise ValueError(
            "Error in _supercell_symgroup_impl: objs tuple cannot be empty"
        )

    for obj in objs:
        if isinstance(obj, clust.Cluster):
            site_indices = {supercell.linear_site_index(s) for s in obj.sites()}
            ops = make_invariant_subgroup(
                configuration=None,
                site_indices=site_indices,
                group=ops,
            )
        elif isinstance(obj, Configuration):
            ops = make_invariant_subgroup(
                configuration=obj,
                group=ops,
            )
        elif isinstance(obj, occ.OccEvent):
            prim = supercell.prim
            std_obj = obj.copy().standardize()
            ops = [
                op
                for op in ops
                if _apply_supercell_symop_to_occ_event(op, obj, prim).standardize()
                == std_obj
            ]
        elif hasattr(obj, "make_invariant_subgroup"):
            ops = obj.make_invariant_subgroup(group=ops)
        else:
            raise TypeError(f"symgroup: unsupported argument type {type(obj).__name__}")

    group = make_symgroup(ops)

    # Determine if last obj is cluster/event
    last_obj = objs[-1]

    if isinstance(last_obj, clust.Cluster):
        prim = supercell.prim
        xtal_prim = prim.xtal_prim

        site_rep = clust.make_integral_site_coordinate_symgroup_rep(
            group.elements, xtal_prim
        )
        group = clust.make_cluster_group(
            cluster=last_obj,
            group=group,
            integral_site_coordinate_symgroup_rep=site_rep,
        )
    elif isinstance(last_obj, occ.OccEvent):
        prim = supercell.prim
        xtal_prim = prim.xtal_prim
        indices = group.head_group_index

        site_rep = clust.make_integral_site_coordinate_symgroup_rep(
            group.elements, xtal_prim
        )
        occ_rep = [prim.occ_symgroup_rep[i] for i in indices]
        atom_rep = [prim.atom_position_symgroup_rep[i] for i in indices]
        occevent_rep = occ.make_occevent_symgroup_rep_from_existing(
            unitcellcoord_symgroup_rep=site_rep,
            occ_symgroup_rep=occ_rep,
            atom_position_symgroup_rep=atom_rep,
        )
        group = occ.make_occevent_group(
            occ_event=last_obj,
            group=group,
            occevent_symgroup_rep=occevent_rep,
        )

    if local is not None:
        if not isinstance(last_obj, (clust.Cluster, occ.OccEvent)):
            raise TypeError(
                f"symgroup local: last obj in objs must be a Cluster or OccEvent "
                f"to use local refinement, got {type(last_obj).__name__}"
            )

        if not isinstance(local, tuple):
            local = (local,)

        # Determine phenomenal cluster/event from last obj — required for local
        prim = supercell.prim
        xtal_prim = prim.xtal_prim

        for obj in local:
            if not isinstance(obj, clust.Cluster):
                raise TypeError(
                    f"symgroup local: expected Cluster, got {type(obj).__name__}"
                )
            site_rep = clust.make_integral_site_coordinate_symgroup_rep(
                group.elements, xtal_prim
            )
            group = clust.make_local_cluster_group(
                cluster=obj,
                phenomenal_group=group,
                integral_site_coordinate_symgroup_rep=site_rep,
            )

    return group


def _supercell_symgroup(self, objs=None, local=None):
    """Return a SymGroup (with supercell lattice) as a subgroup of all
    SupercellSymOp operations for this supercell.

    Parameters
    ----------
    objs : None, object, or tuple[object, ...]
        Specifies what to find the invariant subgroup of, using periodic
        symmetry (symmetry elements must map the objects modulo supercell lattice
        translations):

        - None: return the full SupercellSymOp-based group
        - single object: return the subgroup leaving ``obj`` invariant
        - tuple: sequentially refine

        Supported types:

        - :class:`~libcasm.configuration.Configuration`: uses
          :func:`~libcasm.configuration.make_invariant_subgroup`
        - :class:`~libcasm.clusterography.Cluster`: converts cluster sites
          to linear site indices then uses
          :func:`~libcasm.configuration.make_invariant_subgroup`
        - :class:`~libcasm.occ_events.OccEvent`: filters SupercellSymOp to
          those whose image of the OccEvent equals the original after
          standardization (permutation/reversal), modulo supercell lattice
          translations
        - Any object with a ``make_invariant_subgroup(group: list[SupercellSymOp])``
          method: calls ``obj.make_invariant_subgroup(group=ops)`` to filter the current
          list of SupercellSymOp.

    local : None, Cluster, or tuple[Cluster, ...]
        Optional local cluster refinement applied **after** ``objs``, using
        strict local symmetry (symmetry elements must map the cluster exactly,
        without invoking periodic boundary conditions). Requires the last
        element of ``objs`` to be a :class:`~libcasm.clusterography.Cluster` or
        :class:`~libcasm.occ_events.OccEvent`, which is used as the phenomenal
        cluster/event.

        - None: no local refinement
        - single :class:`~libcasm.clusterography.Cluster`: uses
          :func:`~libcasm.clusterography.make_local_cluster_group`
        - tuple of :class:`~libcasm.clusterography.Cluster`: sequentially
          applies :func:`~libcasm.clusterography.make_local_cluster_group`

    Returns
    -------
    group : libcasm.sym_info.SymGroup
        A SymGroup with the supercell lattice. If the last element of ``objs`` is a
        Cluster or OccEvent, :func:`~libcasm.clusterography.make_cluster_group` or
        :func:`~libcasm.occ_events.make_occevent_group` is called on the SymGroup to
        fix translations so the phenomenal cluster/event maps exactly. If `local` is
        not None, :func:`~libcasm.clusterography.make_local_cluster_group` is applied
        for each local cluster.

    Examples
    --------

    .. code-block:: Python

        # Full SupercellSymOp group
        group = supercell.symgroup()

        # Configuration invariant subgroup
        group = supercell.symgroup(configuration)

        # Cluster invariant subgroup
        group = supercell.symgroup(cluster)

        # Sequential refinements
        group = supercell.symgroup((configuration, cluster))

        # Local cluster group after OccEvent periodic refinement
        group = supercell.symgroup(occ_event, local=local_cluster)

    """
    from ._configuration import make_symgroup

    ops = self.symgroup_rep()

    if objs is not None:
        if not isinstance(objs, tuple):
            objs = (objs,)
        return _supercell_symgroup_impl(ops, objs, self, local=local)

    return make_symgroup(ops)


def _configuration_symgroup(self, objs=None, local=None):
    """Return a SymGroup (with supercell lattice) as a subgroup of all
    SupercellSymOp operations that leave this configuration invariant.

    Parameters
    ----------
    objs : None, object, or tuple[object, ...]
        Specifies additional invariance constraints beyond the configuration
        itself. Same supported types as :func:`Supercell.symgroup`.

    local : None, Cluster, or tuple[Cluster, ...]
        Same as ``local`` in :func:`Supercell.symgroup`.

    Returns
    -------
    group : libcasm.sym_info.SymGroup
        A SymGroup with the supercell lattice. If the last element of ``objs`` is a
        Cluster or OccEvent, :func:`~libcasm.clusterography.make_cluster_group` or
        :func:`~libcasm.occ_events.make_occevent_group` is called on the SymGroup to
        fix translations so the phenomenal cluster/event maps exactly. If `local` is
        not None, :func:`~libcasm.clusterography.make_local_cluster_group` is applied
        for each local cluster.

    Examples
    --------

    .. code-block:: Python

        # Config invariant subgroup
        group = configuration.symgroup()

        # Config ∩ cluster invariant subgroup
        group = configuration.symgroup(cluster)

        # Local cluster group after OccEvent periodic refinement
        group = configuration.symgroup(occ_event, local=local_cluster)

    """
    from ._configuration import make_invariant_subgroup, make_symgroup

    ops = make_invariant_subgroup(configuration=self)

    if objs is not None:
        if not isinstance(objs, tuple):
            objs = (objs,)
        return _supercell_symgroup_impl(ops, objs, self.supercell, local=local)

    return make_symgroup(ops)


def _prim_symgroup(self, objs=None, local=None):
    """Return a SymGroup as a subgroup of Prim.factor_group.

    Parameters
    ----------
    objs : None, object, or tuple[object, ...]
        Specifies what to find the invariant subgroup of, using periodic
        symmetry (symmetry elements must map the objects modulo primitive cell lattice
        translations):

        - None: return ``prim.factor_group``
        - single object: return the subgroup leaving ``obj`` invariant
        - tuple: sequentially refine — for each element, take the subgroup
          of the current group leaving that element invariant

        Supported types:

        - :class:`~libcasm.clusterography.Cluster`: uses
          :func:`~libcasm.clusterography.make_cluster_group`
        - :class:`~libcasm.occ_events.OccEvent`: uses
          :func:`~libcasm.occ_events.make_occevent_group`
        - :class:`~libcasm.xtal.Lattice`: subgroup leaving the lattice
          invariant; use a superlattice of the prim lattice to get a result
          equivalent to :attr:`Supercell.factor_group`

    local : None, Cluster, or tuple[Cluster, ...]
        Optional local cluster refinement applied **after** ``objs``, using
        strict local symmetry (symmetry elements must map the cluster exactly, without
        invoking periodic boundary conditions):

        - None: no local refinement
        - single :class:`~libcasm.clusterography.Cluster`: uses
          :func:`~libcasm.clusterography.make_local_cluster_group`
        - tuple of :class:`~libcasm.clusterography.Cluster`: sequentially
          applies :func:`~libcasm.clusterography.make_local_cluster_group`

        ``local`` is always applied after ``objs``, enforcing the one-way
        transition from periodic to local symmetry.

    Returns
    -------
    group : libcasm.sym_info.SymGroup
        A subgroup of ``prim.factor_group``.

    Examples
    --------

    .. code-block:: Python

        # Factor group
        group = prim.symgroup()

        # Cluster group (periodic — translations allowed)
        group = prim.symgroup(cluster)

        # OccEvent group
        group = prim.symgroup(occ_event)

        # Sequential periodic refinements
        group = prim.symgroup((superlattice, occ_event))

        # Local cluster group: periodic refinement for phenomenal cluster,
        # then local (no-translation) refinement for local cluster
        group = prim.symgroup(phenom_cluster, local=local_cluster)

        # Multiple local refinements after a periodic refinement
        group = prim.symgroup(phenom_cluster, local=(local1, local2))

        # Subgroup leaving a superlattice invariant
        group = prim.symgroup(supercell.superlattice)

    """
    import libcasm.clusterography as clust
    import libcasm.occ_events as occ
    import libcasm.xtal as xtal

    group = self.factor_group

    # --- find invariant subgroup, sequentially for each obj, w/ periodic boundary ---
    if objs is not None:
        if not isinstance(objs, tuple):
            objs = (objs,)

        for obj in objs:
            # head_group_index is [0, 1, ..., n-1] for the factor group itself,
            # and the correct sorted subset for any subgroup thereof.
            indices = group.head_group_index

            if isinstance(obj, clust.Cluster):
                rep = [self.integral_site_coordinate_symgroup_rep[i] for i in indices]
                group = clust.make_cluster_group(
                    cluster=obj,
                    group=group,
                    integral_site_coordinate_symgroup_rep=rep,
                )
            elif isinstance(obj, occ.OccEvent):
                full_rep = occ.make_occevent_symgroup_rep_from_existing(
                    unitcellcoord_symgroup_rep=self.integral_site_coordinate_symgroup_rep,
                    occ_symgroup_rep=self.occ_symgroup_rep,
                    atom_position_symgroup_rep=self.atom_position_symgroup_rep,
                )
                rep = [full_rep[i] for i in indices]
                group = occ.make_occevent_group(
                    occ_event=obj,
                    group=group,
                    occevent_symgroup_rep=rep,
                )
            elif isinstance(obj, xtal.Lattice):
                L = obj.column_vector_matrix()
                L_inv = np.linalg.inv(L)
                invariant = set()
                for i in indices:
                    R = self.factor_group.elements[i].matrix()
                    N = L_inv @ R @ L
                    if np.allclose(N, np.round(N)):
                        invariant.add(i)
                group = group.make_subgroup(invariant)
            else:
                raise TypeError(
                    f"Prim.symgroup: unsupported argument type {type(obj).__name__}"
                )

    # --- local refinements (no periodic boundary conditions) ---
    if local is not None:
        if not isinstance(local, tuple):
            local = (local,)

        for obj in local:
            if not isinstance(obj, clust.Cluster):
                raise TypeError(
                    f"Prim.symgroup local: expected Cluster, got {type(obj).__name__}"
                )
            rep = clust.make_integral_site_coordinate_symgroup_rep(
                group.elements, self.xtal_prim
            )
            group = clust.make_local_cluster_group(
                cluster=obj,
                phenomenal_group=group,
                integral_site_coordinate_symgroup_rep=rep,
            )

    return group
